// 修正後的標準英文四線格與文字繪製邏輯 function drawFourLineGrid(x, y, width, height, textToDraw, isGuide = false) { const lineSpacing = height / 3; // 三個等高格子，共四條線 const line1 = y; // 頂線 (Top line) const line2 = y + lineSpacing; // 中線 (Mid line / Dashed) const line3 = y + lineSpacing * 2; // 基準線 (Baseline / Red) const line4 = y + height; // 底線 (Bottom line) // 1. 繪製四線格 ctx.save(); ctx.lineWidth = 2; // 頂線 (黑色) ctx.strokeStyle = '#2c3e50'; ctx.beginPath(); ctx.moveTo(x, line1); ctx.lineTo(x + width, line1); ctx.stroke(); // 中線 (虛線/藍色) ctx.strokeStyle = '#3498db'; ctx.setLineDash([8, 6]); ctx.beginPath(); ctx.moveTo(x, line2); ctx.lineTo(x + width, line2); ctx.stroke(); ctx.setLineDash([]); // 恢復實線 // 基準線 (紅色/紅粗線) ctx.strokeStyle = '#e74c3c'; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo(x, line3); ctx.lineTo(x + width, line3); ctx.stroke(); // 底線 (黑色) ctx.strokeStyle = '#2c3e50'; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(x, line4); ctx.lineTo(x + width, line4); ctx.stroke(); // 2. 繪製字母 (透光解答或描紅提示) if (textToDraw) { const isUpper = (textToDraw === textToDraw.toUpperCase() && textToDraw.match(/[A-Z]/)); // 文字樣式設定 ctx.textAlign = 'center'; ctx.textBaseline = 'alphabetical'; // 關鍵設定：以英文標準基線對齊 if (isGuide) { ctx.fillStyle = 'rgba(52, 152, 219, 0.25)'; // 描紅或提示虛線 ctx.font = 'bold 110px "Comic Sans MS", "Fredoka", sans-serif'; } else { ctx.fillStyle = 'rgba(46, 204, 113, 0.45)'; // 解答比對透光層 ctx.font = 'bold 110px "Comic Sans MS", "Fredoka", sans-serif'; } // Y 軸對齊點統一設為第三線 (Line 3 - 基準線) // 透過字體與 alphabetical 基線，讓字體自然依據字形特徵落地於第三線 ctx.fillText(textToDraw, x + width / 2, line3); } ctx.restore(); }
